\begin{tikzpicture}
  % The chapter's heart: the weather answers. A question with no owner is
  % asked twice over the boards; the wind crosses with no mouth and barley
  % answers after it, and the row stands where it stood. The two askings lie
  % side by side with a finger's width between them --- two askings are not
  % one witness, and no one stands under them to be asked. Nothing is joined,
  % named, opened, called, spent or ended; the count goes forward.

  % small fleurons at the far ends, quiet bookends
  \node at (-4.55,0.02) {\pgfornament[width=1.0cm,color=ornament,opacity=0.30]{11}};
  \node at ( 4.55,0.02) {\pgfornament[width=1.0cm,color=ornament,symmetry=v,opacity=0.30]{11}};

  % —— the weather: a great wind that crosses the whole with no face, no mouth ——
  \draw[faded!74!ink, line width=0.64pt, opacity=0.44, line cap=round]
    (-4.28,0.78) .. controls (-3.30,1.18) and (-2.36,0.46) .. (-1.14,0.88)
    .. controls (0.06,1.26) and (1.06,0.50) .. (2.28,0.86)
    .. controls (3.28,1.14) and (3.92,0.80) .. (4.32,0.64);

  % a second, fainter gust riding under the first
  \draw[faded!52!ink, line width=0.38pt, opacity=0.26, line cap=round]
    (-4.08,0.50) .. controls (-3.14,0.86) and (-2.40,0.20) .. (-1.20,0.58)
    .. controls (0.00,0.94) and (1.20,0.24) .. (2.40,0.58)
    .. controls (3.30,0.86) and (3.90,0.56) .. (4.26,0.42);

  % the wind goes off up the terrace in a curl, answering and staying unowned
  \draw[faded!58!ink, line width=0.42pt, opacity=0.30, line cap=round]
    (4.32,0.64) .. controls (4.62,0.78) and (4.68,0.44) .. (4.44,0.36)
    .. controls (4.24,0.30) and (4.26,0.54) .. (4.42,0.52);

  % the price of barley answers after it: a small ear riding the wind at right
  \draw[faded!68!ink, line width=0.42pt, opacity=0.34, line cap=round]
    (3.02,1.20) .. controls (3.46,1.34) and (3.94,1.26) .. (4.30,1.06);
  \foreach \bx/\by in {3.16/1.28,3.42/1.33,3.68/1.31,3.92/1.25,4.14/1.16}{
    \fill[gold!60!ornament, opacity=0.52] (\bx,\by) ellipse (0.028 and 0.050);
  }
  \foreach \bx/\by in {3.16/1.28,3.42/1.33,3.68/1.31,3.92/1.25,4.14/1.16}{
    \draw[faded!54!ink, line width=0.26pt, opacity=0.24, line cap=round]
      (\bx,\by) ++(0.01,0.05) -- ++(0.07,0.20);
  }

  % —— the row stands where it stood: four nails and a knot, in their own place ——
  \draw[faded!76!ink, line width=0.58pt, opacity=0.42, line cap=round]
    (-3.10,-0.42) -- (3.10,-0.42);
  \foreach \nx in {-2.50,-1.50,1.50,2.50}{
    \draw[faded!62!ink, line width=0.46pt, opacity=0.34, line cap=round]
      (\nx,-0.49) -- (\nx,-0.35);
  }
  \draw[faded!60!ink, line width=0.44pt, opacity=0.32]
    (0,-0.42) circle (0.055);

  % two askings, side by side, each a hollow curl with no mouth and no owner
  \draw[faded!66!ink, line width=0.40pt, opacity=0.36, line cap=round]
    (-1.06,0.12) .. controls (-1.24,0.34) and (-0.82,0.40) .. (-0.84,0.12)
    .. controls (-0.86,-0.06) and (-1.06,-0.04) .. (-1.00,0.12);
  \draw[faded!66!ink, line width=0.40pt, opacity=0.36, line cap=round]
    ( 0.86,0.12) .. controls ( 0.68,0.34) and ( 1.10,0.40) .. ( 1.08,0.12)
    .. controls ( 1.06,-0.06) and ( 0.86,-0.04) .. ( 0.92,0.12);
  \fill[faded!56!ink, opacity=0.30] (-0.95,-0.13) circle (0.020);
  \fill[faded!56!ink, opacity=0.30] ( 0.95,-0.13) circle (0.020);

  % the finger's width held between them: two askings are not one witness
  \draw[faded!58!ink, line width=0.44pt, opacity=0.30, line cap=round]
    (0,-0.30) -- (0,0.20);
  \draw[faded!50!ink, line width=0.34pt, opacity=0.22, line cap=round]
    (-0.05,0.20) .. controls (0,0.28) and (0,0.28) .. (0.05,0.20);

  \node[font=\FellFont\itshape\small, text=faded] at (0,-1.30)
    {the weather answers; the row stands};
\end{tikzpicture}
