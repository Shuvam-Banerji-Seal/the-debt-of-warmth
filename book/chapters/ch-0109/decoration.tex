  % The chapter's heart: eleven arithmetic cuts stand around a single figure
  % and never close into an account. The figure is read and not paid; the space
  % beneath it carries no signature and no hand that would set one there. The
  % slate does not know it has been read; the house keeps the count in loaves.

  % —— two small flourons, far apart, the only ornament above the slab ——
  \node at (-4.55,0.72) {\pgfornament[width=0.46cm,color=ornament,opacity=0.24]{30}};
  \node at ( 4.55,0.72) {\pgfornament[width=0.46cm,color=ornament,opacity=0.24,symmetry=v]{30}};

    (-3.78,1.00) -- (3.54,1.00) -- (3.68,0.80) -- (3.72,-0.98)
    -- (-3.70,-1.02) -- (-3.84,0.78) -- cycle;
  % the cold light along the top, the dark along the left
  \draw[ash!52!vellum, line width=0.55pt, opacity=0.42, line cap=round]
    (-3.74,0.96) -- (3.50,0.96);
  \draw[night!60!ash, line width=0.40pt, opacity=0.40, line cap=round]
    (-3.80,-0.96) -- (-3.80,0.74);

  % —— eleven cuts in the slate, grouped five and five and one ——
  % each leans toward the figure and stops short of it; none signs, none joins

        (0.035,0.14) -- (0.035,-0.14);
