\begin{tikzpicture}
  % The chapter's heart: Frost's second arithmetic. On the Account-day a
  % thane read one in twelve twice over the basket, once for the moot and
  % once for the late households, called the levy and denied it in the same
  % breath, taking himself out of his own sentence as a surgeon withdraws a
  % hand from a field before the dressing goes on. The subtraction went up
  % the road locked and cannot be changed back. The same figure goes over
  % the basket twice; the figure read is still not the figure paid. At the
  % right a slate goes up the road and comes back with a figure nobody here
  % read, carried and unopened.

  % —— the locked box: the subtraction up the road, not changed back ——

    (-4.10,-0.30) rectangle (-2.62,0.42);

    (-4.10,-0.30) rectangle (-2.62,0.42);

    (-4.10,0.20) -- (-2.62,0.20);
  \foreach \mx in {-3.86,-3.52,-3.18,-2.84}{
    \draw[ink!56, line width=0.46pt, opacity=0.42, line cap=round]
      (\mx,-0.14) -- (\mx,0.08);
  }
  \fill[ornament!62!ink, opacity=0.52] (-3.36,0.20) circle (0.055);
  \draw[ink!64, line width=0.40pt, opacity=0.50, line cap=round]
    (-3.36,0.20) -- (-3.36,0.06);

  % —— the basket, set in shade, its opening kept dark ——
