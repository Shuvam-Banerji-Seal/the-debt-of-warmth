\begin{tikzpicture}
  % the chapter's heart: the eighth word brought to the teeth and gone back.
  % One word-object is held in the gate between the jaws --- an interval with
  % two ends, not a payment, no figure and no name. Eight ripples go out from
  % it and turn back, each broken at a different place, so the eight comings
  % are nested about the word rather than laid side by side into an account:
  % they cannot stand up together and call themselves a doctrine, for a
  % doctrine is an account that has found a payer, and none is named here.
  % The pale sliver behind is the reed month going out, the account passing
  % into the next house unread.

  % the month going out: a waning sliver, low and wide
  \fill[gold, opacity=0.032]
    (-4.90,0.10) .. controls (-2.0,-0.56) and (2.0,-0.56) .. (4.90,0.10)
    .. controls (2.0,0.44) and (-2.0,0.44) .. (-4.90,0.10) -- cycle;

  % eight comings and goings, nested and broken, each turning back to the
  % word and never passing outward into a hand. The innermost is the eighth.
  \foreach \rx/\ry/\ang/\op/\col in {
      0.62/0.27/0/0.52/rubric!48!faded,
      1.02/0.42/45/0.44/faded!78!ink,
      1.42/0.57/90/0.38/faded!78!ink,
      1.82/0.72/135/0.32/faded!78!ink,
      2.22/0.87/180/0.27/faded!78!ink,
      2.62/1.02/225/0.22/faded!78!ink,
      3.02/1.17/270/0.18/faded!78!ink,
      3.42/1.32/315/0.14/faded!78!ink}{
    \pgfmathsetmacro{\astart}{\ang+14}
    \pgfmathsetmacro{\aend}{\ang+346}
    \draw[\col, line width=0.55pt, opacity=\op]
      plot[domain=\astart:\aend, samples=44]
      ({0.025+(\rx+0.012*sin(3*\x))*cos(\x)},
       {0.025+(\ry+0.010*cos(4*\x))*sin(\x)});
  }

  % the jaws: two shallow curves that do not quite close
  \draw[faded!82!ink, line width=0.92pt, opacity=0.58]
    plot[domain=-1.52:1.52, samples=40] (\x, {0.24+0.34*(\x/1.5)*(\x/1.5)});
  \draw[faded!82!ink, line width=0.92pt, opacity=0.58]
    plot[domain=-1.52:1.52, samples=40] (\x, {-0.16-0.32*(\x/1.5)*(\x/1.5)});
  \draw[faded!78!ink, line width=0.75pt, opacity=0.42]
    (1.50,0.58) .. controls (1.70,0.30) and (1.70,-0.18) .. (1.50,-0.48);
  \draw[faded!78!ink, line width=0.75pt, opacity=0.42]
    (-1.50,0.58) .. controls (-1.70,0.30) and (-1.70,-0.18) .. (-1.50,-0.48);

  % the teeth, hanging above and rising below; none at the centre, where the
  % gate is left open for the word
  \foreach \tx in {-1.30,-1.05,-0.80,-0.55,-0.30,0.30,0.55,0.80,1.05,1.30}{
    \pgfmathsetmacro{\uty}{0.24+0.34*(\tx/1.5)*(\tx/1.5)}
    \pgfmathsetmacro{\lty}{-0.16-0.32*(\tx/1.5)*(\tx/1.5)}
    \fill[faded!16!vellum, opacity=0.55]
      (\tx-0.044,\uty) -- (\tx+0.044,\uty) -- (\tx,\uty-0.135) -- cycle;
    \draw[faded!76!ink, line width=0.45pt, opacity=0.42]
      (\tx-0.044,\uty) -- (\tx,\uty-0.135) -- (\tx+0.044,\uty);
    \fill[faded!16!vellum, opacity=0.55]
      (\tx-0.044,\lty) -- (\tx+0.044,\lty) -- (\tx,\lty+0.125) -- cycle;
    \draw[faded!76!ink, line width=0.45pt, opacity=0.42]
      (\tx-0.044,\lty) -- (\tx,\lty+0.125) -- (\tx+0.044,\lty);
  }

  % the word held at the teeth: a measured interval with two ends, not paid
  \fill[rubric!45!faded, opacity=0.30] (-0.25,-0.025) rectangle (0.25,0.075);
  \draw[rubric!82!ink, line width=1.15pt, opacity=0.82] (-0.25,0.025) -- (0.25,0.025);
  \draw[rubric!76!ink, line width=0.95pt, opacity=0.70] (-0.25,-0.025) -- (-0.25,0.075);
  \draw[rubric!76!ink, line width=0.95pt, opacity=0.70] (0.25,-0.025) -- (0.25,0.075);
  \fill[gold, opacity=0.55] (0,0.025) circle (0.028);

  % the reach toward the teeth, stopped short with no figure passing through
  \draw[faded, line width=0.50pt, dotted, opacity=0.32] (0,0.078) -- (0,0.112);
  \draw[rubric!55!faded, line width=0.80pt, opacity=0.55] (-0.055,0.112) -- (0.055,0.112);

  \node at (-4.78,0.02) {\pgfornament[width=0.95cm,color=ornament]{94}};
  \node at ( 4.78,0.02) {\pgfornament[width=0.95cm,color=ornament,symmetry=v]{94}};

  \node[font=\FellFont\itshape\small, text=faded] at (0,-1.46)
    {eight at the teeth, and no payer named};
\end{tikzpicture}
