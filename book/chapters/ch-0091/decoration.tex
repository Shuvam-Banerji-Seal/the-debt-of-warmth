  % Six notches stand in the worn mark where the house can check; the seventh
  % sits at the lip, beyond checking, and is not cut. The knife was set where
  % it could be checked and also slipped off; the thumb came back to six.
  % The touch took nothing, paid nothing, and owed nothing --- one object on
  % a place of its own, kept because it was true and not because it bought.

  % —— a small fleuron above ——
  \node at (0,1.52) {\pgfornament[width=0.72cm,color=ornament,opacity=0.24]{6}};

    -- cycle;
  \draw[ornament!44!ink, line width=0.40pt, opacity=0.42, line cap=round]
    (-2.96,0.04) .. controls (-3.14,0.26) and (-3.12,0.48) .. (-2.92,0.56)
    -- (2.62,0.56) .. controls (2.80,0.52) and (2.84,0.28) .. (2.72,0.04);
  \draw[ink!46!faded, line width=0.30pt, opacity=0.38, line cap=round]
    (-2.92,0.56) -- (2.62,0.56);
  % two faint lines of grain
  \draw[faded!25!ink, line width=0.22pt, opacity=0.24, line cap=round]
    (-2.60,0.34) .. controls (-1.20,0.30) and (0.80,0.38) .. (2.30,0.32);
