\begin{tikzpicture}
  % The chapter's heart: the figure without a signature. Frost's third
  % reading changes the district's arithmetic a third time and cannot be
  % changed back. The figure is read twice and stands unchallenged; the
  % price stands where nobody set it down. Between them the line for a
  % signature is left empty --- no payer, no witness, no hand, no name.
  % Along the top, three readings stand on three places with a finger's
  % width between: three, and not one account.

  % —— a small fleur above the three readings ——
  \node at (0,1.62) {\pgfornament[width=0.9cm,color=ornament,opacity=0.24]{6}};

  % —— three readings on three separate places, with a finger between ——
  \foreach \cx in {-2.20,0.00,2.20}{
    \fill[faded!7!vellum, opacity=0.82] (\cx-0.40,0.98) rectangle (\cx+0.40,1.10);
    \draw[faded!38!ink, line width=0.36pt, opacity=0.32]
      (\cx-0.40,0.98) rectangle (\cx+0.40,1.10);
  }
  \foreach \cx in {-2.20,0.00}{
    \draw[ink!58!faded, line width=0.82pt, opacity=0.42, line cap=round]
      (\cx-0.09,1.14) -- (\cx-0.09,1.34);
    \draw[ink!58!faded, line width=0.82pt, opacity=0.42, line cap=round]
      (\cx+0.09,1.14) -- (\cx+0.09,1.34);
  }
  % the third subtraction: crossed once, and not to be crossed back
  \draw[rubric!74!ink, line width=0.90pt, opacity=0.74, line cap=round]
    (2.11,1.14) -- (2.11,1.34);
  \draw[rubric!74!ink, line width=0.90pt, opacity=0.74, line cap=round]
    (2.29,1.14) -- (2.29,1.34);
  \draw[rubric!74!ink, line width=0.72pt, opacity=0.68, line cap=round]
    (2.03,1.24) -- (2.37,1.24);
  \foreach \gx in {-1.10,1.10}{
    \draw[faded!30!ink, line width=0.24pt, opacity=0.22,
          dash pattern=on 1pt off 2.2pt] (\gx,0.96) -- (\gx,1.38);
  }
  \node[font=\CinzelFont\scriptsize, text=faded, opacity=0.48] at (0,0.82)
    {three readings, not one account};

  % —— the figure, printed and unchallenged ——
  \fill[faded!10!vellum, opacity=0.88] (-3.95,-0.34) rectangle (-1.75,0.58);
  \draw[ornament!52!ink, line width=0.46pt, opacity=0.42]
    (-3.95,-0.34) rectangle (-1.75,0.58);
  \draw[faded!32!ink, line width=0.30pt, opacity=0.28]
    (-3.82,0.42) -- (-1.88,0.42);
  \node[font=\CinzelFont\small, text=rubric, opacity=0.80] at (-2.85,0.12)
    {one in twelve};
  \draw[rubric!80!ink, line width=1.5pt, opacity=0.68, line cap=round]
    (-3.30,-0.06) -- (-2.40,-0.06);
  \node[font=\FellFont\itshape\scriptsize, text=faded, opacity=0.52] at (-2.85,-0.62)
    {read twice, unchallenged};

  % —— the price, standing where nobody set it down ——
  \fill[ash!8!vellum, opacity=0.84] (1.75,-0.34) rectangle (3.95,0.58);
  \draw[faded!40!ink, line width=0.44pt, opacity=0.36]
    (1.75,-0.34) rectangle (3.95,0.58);
  \draw[faded!30!ink, line width=0.30pt, opacity=0.26]
    (1.88,0.42) -- (3.82,0.42);
  \node[font=\CinzelFont\small, text=ink!70!faded, opacity=0.64] at (2.85,0.12)
    {the price};
  \draw[ink!62!faded, line width=1.2pt, opacity=0.46, line cap=round]
    (2.40,-0.06) -- (3.30,-0.06);
  \node[font=\FellFont\itshape\scriptsize, text=faded, opacity=0.52] at (2.85,-0.62)
    {paid, with no name};

  % —— the empty rule where a signature would stand: nothing written ——
  \draw[faded!46!ink, line width=0.34pt, opacity=0.34,
        dash pattern=on 1pt off 2.4pt] (-1.28,-0.16) -- (1.28,-0.16);
  \foreach \tx in {-1.28,1.28}{
    \draw[ink!56, line width=0.34pt, opacity=0.40, line cap=round]
      (\tx,-0.28) -- (\tx,-0.04);
  }
  \node[font=\FellFont\itshape\scriptsize, text=rubric, opacity=0.60] at (0,0.10)
    {no signature stands};
  \node[font=\CinzelFont\scriptsize, text=faded, opacity=0.48] at (0,-0.62)
    {figure read, not the figure paid};

  % —— the finding kept on a place of its own, and the motto ——
  \node[font=\FellFont\itshape\small, text=faded] at (0,-1.02)
    {the price stands where nobody set it down};
  \node at (0,-1.44) {\pgfornament[width=4.6cm,color=ornament,opacity=0.22,symmetry=v]{101}};
\end{tikzpicture}
