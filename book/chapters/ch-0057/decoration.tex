\begin{tikzpicture}
  % The chapter's heart: the strips give back their first grain, set under
  % wool and counted in glances and handfuls, entered on no slate and given
  % no figure. The yard goes around an empty place in the returning line
  % with a look and a wide step and no number. Bundle and absence lie on
  % separate trays and are joined to no account.

  % the first grain, come back: a wheat sprig held over the whole, not a crown
  \node at (0,1.46) {\pgfornament[width=1.10cm,color=ornament,opacity=0.52]{94}};

  % the returning line: it comes in from the left, widens around a blank
  % place, and goes on to the right. Two branches leave the same point and
  % meet again; the space between them is left empty on purpose.
  \draw[faded!84!ink, line width=0.72pt, opacity=0.54, line cap=round]
    (-4.02,0.34) .. controls (-3.10,0.36) and (-2.20,0.38) .. (-1.45,0.40);
  \draw[faded!84!ink, line width=0.72pt, opacity=0.54, line cap=round]
    ( 1.45,0.40) .. controls (2.20,0.38) and (3.10,0.36) .. (4.02,0.34);
  \draw[faded!76!ink, line width=0.60pt, opacity=0.44, line cap=round]
    (-1.45,0.40) .. controls (-1.12,1.02) and (1.12,1.02) .. (1.45,0.40);
  \draw[faded!76!ink, line width=0.60pt, opacity=0.44, line cap=round]
    (-1.45,0.40) .. controls (-1.12,-0.30) and (1.12,-0.30) .. (1.45,0.40);

  % a fainter second going-around, close under the first, so the place is
  % passed once more and still contains nothing
  \draw[faded!46!ink, line width=0.32pt, opacity=0.22, line cap=round]
    (-1.30,0.32) .. controls (-1.04,-0.20) and (1.04,-0.20) .. (1.30,0.32);

  % a look: two small marks on the upper branch, glances that keep their
  % own count and are not entered
  \fill[faded!70!ink, opacity=0.44] (-0.86,0.70) circle (0.022);
  \fill[faded!70!ink, opacity=0.44] ( 0.86,0.70) circle (0.022);

  % the wide step at the ends of the detour: two short outward ticks
  \draw[faded!62!ink, line width=0.40pt, opacity=0.34, line cap=round]
    (-1.45,0.52) -- (-1.58,0.60);
  \draw[faded!62!ink, line width=0.40pt, opacity=0.34, line cap=round]
    ( 1.45,0.52) -- ( 1.58,0.60);

  % set near the hearth and covered with wool: a small sheaf under a fold,
  % its stalks showing, tied and unfigured
  \draw[faded!80!ink, line width=0.52pt, opacity=0.44, line cap=round]
    (-2.70,-0.46) .. controls (-2.82,-0.28) and (-2.86,-0.14) .. (-2.90,-0.04);
  \draw[faded!72!ink, line width=0.48pt, opacity=0.38, line cap=round]
    (-2.70,-0.46) .. controls (-2.72,-0.24) and (-2.70,-0.12) .. (-2.68,0.00);
  \draw[faded!64!ink, line width=0.44pt, opacity=0.32, line cap=round]
    (-2.70,-0.46) .. controls (-2.58,-0.28) and (-2.54,-0.16) .. (-2.50,-0.06);
  \foreach \gp in {(-2.90,-0.04),(-2.86,-0.15),(-2.68,0.00),(-2.50,-0.06),(-2.54,-0.17)}{
    \fill[gold!55!ornament, opacity=0.42] \gp ellipse (0.026 and 0.052);
  }
  \draw[rubric!45!faded, line width=0.54pt, opacity=0.34, line cap=round]
    (-2.84,-0.36) .. controls (-2.70,-0.32) and (-2.60,-0.34) .. (-2.56,-0.38);
  \filldraw[fill=faded!12!vellum, fill opacity=0.86,
            draw=faded!72!ink, draw opacity=0.42, line width=0.52pt]
    (-3.02,-0.78)
    .. controls (-2.94,-0.52) and (-2.86,-0.48) .. (-2.70,-0.50)
    .. controls (-2.52,-0.52) and (-2.44,-0.58) .. (-2.38,-0.76)
    .. controls (-2.60,-0.88) and (-2.82,-0.88) .. cycle;

  % handfuls: a few flecks beside the sheaf, known to the arm, not the slate
  \foreach \d in {(-3.20,-0.30),(-3.12,-0.20),(-2.28,-0.26)}{
    \fill[ink!42!faded, opacity=0.30] \d circle (0.015);
  }

  % the ends of the returning line go on without a figure or a stopping
  \fill[faded!58!ink, opacity=0.32] (-4.02,0.34) circle (0.020);
  \fill[faded!58!ink, opacity=0.32] ( 4.02,0.34) circle (0.020);

  \node[font=\FellFont\itshape\small, text=faded] at (0,-1.46)
    {the yard goes around, and gives no number};
\end{tikzpicture}
