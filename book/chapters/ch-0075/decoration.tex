  % the well lip wrote did not go when the sun was up, as they had gone on
  % Frost 4, and the yard still called it weather. On Frost 9 the trough
  % skinned from edge to edge; Sella pressed the back of the broken pot to the
  % skin and lifted a plate of ice clouded with silt, set it on a flat stone
  % out of the path of feet, and told the weather where to stand. Nobody gave
  % it a figure. A vessel with a skin laid over it, and a faint sun that
  % cannot take its writing back.

  % —— the skinned trough: a vessel with a skin laid over it ——

    (-3.55,-0.52) .. controls (-2.55,-0.66) and (-1.55,-0.66) .. (-1.48,-0.52);

    (-2.52,-0.10) ellipse (1.08 and 0.28);
  \draw[ornament!56!ink, line width=0.56pt, opacity=0.52]
    (-2.52,-0.10) ellipse (1.08 and 0.28);
  \draw[faded!34!ink, line width=0.26pt, opacity=0.22, line cap=round]
    (-3.36,-0.16) .. controls (-2.80,-0.03) and (-2.16,-0.18) .. (-1.72,-0.08);
