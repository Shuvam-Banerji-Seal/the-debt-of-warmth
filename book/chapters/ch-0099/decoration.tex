\begin{tikzpicture}
  % The chapter's heart: eight arithmetics go up the clay road locked and
  % unread in one box, and one figure stands on the notice. Eight are eight
  % and not one account; the figure read is still not the figure paid, and
  % no hand writes on the space between the box and the board. What the
  % notice does not know is the eight that stand behind its single mark.

  % —— a small fleuron above ——
  \node at (0,1.46) {\pgfornament[width=0.70cm,color=ornament,opacity=0.24]{11}};

  % —— eight slats in a bound bundle, locked and unread ——
  \foreach \k in {0,1,2,3,4,5,6,7}{
    \pgfmathsetmacro{\sy}{-0.15 + 0.14*\k + (\k>3 ? 0.08 : 0)}
    \draw[ink!46!faded, line width=1.1pt, opacity=0.54, line cap=round]

    \draw[faded!26!ink, line width=0.8pt, opacity=0.44, line cap=round]
      (-2.63,\sy-0.06) -- (-2.63,\sy+0.06);
  }
  \foreach \cx in {-4.25,-2.95}{
    \draw[ornament!44!ink, line width=0.7pt, opacity=0.50, line cap=round]
      (\cx,-0.24) -- (\cx,0.96);
    \draw[ornament!44!ink, line width=0.7pt, opacity=0.40, line cap=round]
      (\cx-0.05,0.92) -- (\cx+0.05,0.92);
  }
  \node[font=\CinzelFont\scriptsize, text=faded, opacity=0.42] at (-3.70,-0.44)
    {eight, locked and unread};

  % —— the box: strapped, hasped, and padlocked ——
  \fill[vellum, opacity=0.46]
    (-2.35,-0.62) rectangle (-0.15,0.20);
  \draw[ornament!42!ink, line width=0.42pt, opacity=0.52, line cap=round]

  \node[font=\FellFont\itshape\scriptsize, text=faded, opacity=0.46] at (0.91,0.22)
    {no hand writes between};

  \foreach \i in {0,1,2,3,4,5,6,7}{
