\begin{tikzpicture}
  % the chapter's heart: the hand that goes BENEATH the loaf shelf.
  % The board carries its row and its two unnamed gaps and says nothing;
  % under the front edge the charcoal lies waiting, unread and with no
  % figure. A finger comes up from underneath, holds for the length of
  % one counted breath, and is taken away without a look. The beat is
  % filed on a place of its own --- a small closed ring the dotted line
  % approaches but never joins --- held and unspoken, not a reading,
  % not a payment, not made a pair with the other fractions into a rule.
  % The road knows the weight it is not asked to read.

  % the board, drawn long and low, its front edge the line the hand knows
  \fill[faded!12!vellum, opacity=0.62]
    (-3.85,0.12) -- (3.85,0.12) -- (3.68,0.47) -- (-3.68,0.47) -- cycle;
  \draw[faded!82!ink, line width=0.92pt, opacity=0.60]
    (-3.85,0.12) -- (3.85,0.12) -- (3.68,0.47) -- (-3.68,0.47) -- cycle;
  \draw[faded!60!ink, line width=0.44pt, opacity=0.32]
    (-3.62,0.23) -- (3.62,0.23);
  \draw[faded!48!ink, line width=0.40pt, opacity=0.24]
    (-3.30,0.38) -- (3.32,0.38);

  % the row above, suggested in low arcs; two spaces stay where they are,
  % the old place at the row's end and the palm's breadth, unnamed
  \foreach \lx in {-3.05,-2.24,-1.43,0.62,1.43,2.24,3.05}{
    \draw[faded!72!ink, line width=0.62pt, opacity=0.34]
      (\lx-0.28,0.48) .. controls (\lx,0.86) and (\lx+0.02,0.86) .. (\lx+0.28,0.48);

  \draw[faded, line width=0.40pt, dotted, opacity=0.26] (-0.98,0.50) -- (-0.28,0.50);
  \draw[faded, line width=0.40pt, dotted, opacity=0.26] (0.00,0.50) -- (0.30,0.50);

  % the charcoal tally beneath the front edge: short uneven marks,
  % none of them gathered into a figure, waiting for a hand that is not his

      (\cx,0.01) -- (\cx+\cl,0.13);
  }

  % the hand from underneath --- not a reach from the road, but a finger
  % rising under the board and resting against its underside

    (-0.68,0.00) -- (-0.68,-0.22);

  % one counted breath, held: the dotted line comes near and stops short,
  % and the small closed ring is kept on a place of its own
